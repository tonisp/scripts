// Normalize value to range [-1, 1]
normalize(float val, float minV, float maxV) =>
    math.max(-1.0, math.min(1.0, (val - (minV + maxV) / 2) / ((maxV - minV) / 2)))

// Safe number string helper (prevents NaN/crash)
safeNum(float val, string fmt) =>
    na(val) ? "N/A" : str.tostring(val, fmt)

// Auto Higher Timeframe Resolution
getHTF() =>
    if timeframe.isseconds or (timeframe.isintraday and timeframe.multiplier <= 5)
        "60"
    else if timeframe.isintraday and timeframe.multiplier <= 15
        "240"
    else if timeframe.isintraday and timeframe.multiplier <= 60
        "D"
    else if timeframe.isintraday
        "W"
    else if timeframe.isdaily
        "W"
    else if timeframe.isweekly
        "M"
    else
        "M"

htfRes = getHTF()

// Color palette function for table and visuals
scoreColor(float sc) =>
    na(sc) ? color.gray : sc >= 0.5 ? color.new(#00c853, 0) : sc >= 0.2 ? color.new(#69f0ae, 0) : sc >= -0.2 ? color.new(#ffd54f, 0) : sc >= -0.5 ? color.new(#ff8a65, 0) : color.new(#ff1744, 0)

// Choppiness Index (CHOP) - detects consolidation vs trending
chopSumAtr = math.sum(ta.atr(1), chopLen)
chopHiLo   = ta.highest(high, chopLen) - ta.lowest(low, chopLen)
chopVal    = chopHiLo > 0 ? 100 * (math.log10(chopSumAtr / chopHiLo) / math.log10(chopLen)) : 50.0
isChoppy   = useChopFilter and (chopVal >= chopThresh)


// ══════════════════════════════════════════════════════════════════════════════
// ░░░░░░░░░░░░░░░░░░░░░░░  1. TREND & CHOP FILTERS  ░░░░░░░░░░░░░░░░░░░░░░░░░░
// ══════════════════════════════════════════════════════════════════════════════

// --- Golden & Death Cross Benchmark (TradingAgents 50 vs 200) ---
sma50       = ta.sma(close, emaMid)
sma200      = ta.sma(close, emaSlow)
goldenCross = sma50 > sma200

crossScore = 0.0
if useGoldenCross
    crossScore := goldenCross ? 0.35 : -0.35

// --- EMA ---
ema21  = ta.ema(close, emaFast)
ema50  = ta.ema(close, emaMid)
ema200 = ta.ema(close, emaSlow)

emaScore = 0.0
emaScore := close > ema200 ? emaScore + 0.30 : emaScore - 0.30
emaScore := close > ema50  ? emaScore + 0.30 : emaScore - 0.30
emaScore := ema21 > ema50  ? emaScore + 0.25 : emaScore - 0.25
if useGoldenCross
    emaScore := emaScore + crossScore * 0.15
emaScore := math.max(-1.0, math.min(1.0, emaScore))

// --- ADX / DMI ---
[diPlus, diMinus, adxVal] = ta.dmi(adxLen, adxLen)
adxScore = 0.0
trendStrength = adxVal > adxThresh
adxScore := trendStrength ? (diPlus > diMinus ? 1.0 : -1.0) : 0.0

// --- Supertrend ---
[stLine, stDir] = ta.supertrend(stFactor, stPeriod)
stScore = stDir == -1 ? 1.0 : -1.0

// --- UT Bot Alerts ---

utHaTicker = ticker.heikinashi(syminfo.tickerid)

          utSrc < nz(utStop[1], 0) and utSrc[1] < nz(utStop[1], 0) ? math.min(nz(utStop[1]), utSrc + utLoss) : 
          utSrc > nz(utStop[1], 0) ? utSrc - utLoss : utSrc + utLoss
          
var int utPos = 0   
utPos := utSrc[1] < nz(utStop[1], 0) and utSrc > nz(utStop[1], 0) ? 1 :
         utSrc[1] > nz(utStop[1], 0) and utSrc < nz(utStop[1], 0) ? -1 : nz(utPos[1], 0)

// --- Ichimoku Cloud ---
tenkan   = (ta.highest(high, 9) + ta.lowest(low, 9)) / 2
kijun    = (ta.highest(high, 26) + ta.lowest(low, 26)) / 2
senkouA  = (tenkan + kijun) / 2

ichiScore := close > tenkan ? ichiScore + 0.25 : ichiScore - 0.25

var float lastSwingLow  = na

smcPh = ta.pivothigh(high, smcLen, smcLen)

macdCrossDown = ta.crossunder(macdLine, signalLine)

var float prevPhRsi   = na

sqzBearDecel = sqzVal < 0 and sqzValRising

notContradictSell  = not (trendScore > 0.5 and momScore > 0.5)
regimeAllowed      = not (useChopFilter and chopVal >= effChopThresh)
ema200ConfirmBuy   = not effRequireEma200 or close > ema200
ema200ConfirmSell  = not effRequireEma200 or close < ema200
rsiFilterBuy       = not useRsiFilter or rsiVal <= rsiMaxBuyLevel
rsiFilterSell      = not useRsiFilter or rsiVal >= rsiMinSellLevel
macdHistDropBuy    = effBlockHistDrop and macdLine > 0 and histLine < histLine[1] and histLine[1] < histLine[2]
macdOverextendedBuy = macdLine > atrVal * macdMaxAtrMult
macdFilterBuy      = not useMacdFilter or not (macdHistDropBuy or macdOverextendedBuy)
macdHistRiseSell   = effBlockHistDrop and macdLine < 0 and histLine > histLine[1] and histLine[1] > histLine[2]
macdOverextendedSell = macdLine < -atrVal * macdMaxAtrMult
macdFilterSell     = not useMacdFilter or not (macdHistRiseSell or macdOverextendedSell)
smcFilterBuy       = not effRequireBOS or marketStructure >= 0
smcFilterSell      = not effRequireBOS or marketStructure <= 0

utBotConfirmSell   = not requireUtBot or utPos == -1

// --- Standard Trend-Following Signals (Ideal for 30m / 1h) ---
standardBuy  = compositeScore >= effBuyThresh and barsSinceSignal >= effCooldown and volumeConfirm and notContradictBuy and regimeAllowed and paConfirmBuy and inSession and ema200ConfirmBuy and rsiFilterBuy and macdFilterBuy and smcFilterBuy and utBotConfirmBuy
standardSell = compositeScore <= effSellThresh and barsSinceSignal >= effCooldown and volumeConfirm and notContradictSell and regimeAllowed and paConfirmSell and inSession and ema200ConfirmSell and rsiFilterSell and macdFilterSell and smcFilterSell and utBotConfirmSell

crossEma21Up   = ta.crossover(close, ema21)
crossEma21Down = ta.crossunder(close, ema21)
crossOpenUp    = ta.crossover(close, open)
crossOpenDown  = ta.crossunder(close, open)
crossStochUp   = ta.crossover(stochK, stochD)
crossStochDown = ta.crossunder(stochK, stochD)
// --- Fibonacci Calculations (Dynamic Lookback) ---
macroHigh = ta.highest(high, effFibLookback)
macroLow  = ta.lowest(low, effFibLookback)
fibRange  = macroHigh - macroLow
fib236    = macroHigh - fibRange * 0.236
fib382    = macroHigh - fibRange * 0.382
fib500    = macroHigh - fibRange * 0.500
fib618    = macroHigh - fibRange * 0.618
fib786    = macroHigh - fibRange * 0.786

fibFilterBuy  = useFibFilter ? (close <= fib500) : true
fibFilterSell = useFibFilter ? (close >= fib500) : true

// --- Early Reversal Signals (Point 2 B - Ideal for 1m / 5m / 15m Scalping) ---
earlyDivTriggerBuy    = useDivTrigger and (bullishDiv or (bullDivActive and crossEma21Up))
earlySqzTriggerBuy    = useSqzFireTrigger and (sqzFiredUp or (sqzBearDecel and crossOpenUp))
earlyStochTriggerBuy  = useStochTrigger and (crossStochUp and stochK < 40)
earlyBuyCondition     = earlyDivTriggerBuy or earlySqzTriggerBuy or earlyStochTriggerBuy
earlyBuySafety        = (not requireEma50Early or close > ema50) and regimeAllowed and notContradictBuy and inSession and (close >= open or close > close[1]) and rsiFilterBuy and fibFilterBuy
earlyBuySignal        = effEarlyReversal and earlyBuyCondition and earlyBuySafety and compositeScore >= effEarlyThresh and barsSinceSignal >= effCooldown and utBotConfirmBuy

earlyDivTriggerSell   = useDivTrigger and (bearishDiv or (bearDivActive and crossEma21Down))
// Squeeze dark green requires either an EMA21 breakdown OR a valid strict bearish divergence
earlySqzTriggerSell   = useSqzFireTrigger and (sqzFiredDown or (sqzBullDecel and crossEma21Down) or (sqzBullDecel and bearishDiv))
earlyStochTriggerSell = useStochTrigger and (crossStochDown and stochK > 60)
earlySellCondition    = earlyDivTriggerSell or earlySqzTriggerSell or earlyStochTriggerSell
earlySellSafety       = (not requireEma50Early or close < ema50) and regimeAllowed and notContradictSell and inSession and (close <= open or close < close[1]) and rsiFilterSell and fibFilterSell
earlySellSignal       = effEarlyReversal and earlySellCondition and earlySellSafety and compositeScore <= -effEarlyThresh and barsSinceSignal >= effCooldown and utBotConfirmSell
